package main

import (
	"context"
	"encoding/json"
	"fmt"
	"test-mongo/db"

	"go.mongodb.org/mongo-driver/v2/bson"
	"go.mongodb.org/mongo-driver/v2/mongo"
)

type Word struct {
	Sets []string
}

func main() {
	var dbClient *mongo.Client = db.DbConnect()

	var coll = dbClient.Database("tantei").Collection("words")
	var value string = "tabemono"

	var result bson.M
	var err = coll.FindOne(context.TODO(), bson.D{{"latin", value}}).Decode(&result)

	if err == mongo.ErrNoDocuments {
		fmt.Printf("No document was found with the title %s\n", value)
		return
	}
	if err != nil {
		panic(err)
	}

	jsonData, err := json.MarshalIndent(result, "", "    ")
	if err != nil {
		panic(err)
	}

	var w Word
	err = json.Unmarshal(jsonData, &w)

	fmt.Printf("%s\n", jsonData)
	fmt.Printf("%v\n", w.Sets[0])
}
