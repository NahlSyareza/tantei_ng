package models

import (
	"backpoint-partners/db"
	"time"

	"go.mongodb.org/mongo-driver/v2/bson"
	"go.mongodb.org/mongo-driver/v2/mongo"
)

type WordSchema struct {
	Id          bson.ObjectID `bson:"_id"`
	Japanese    string        `bson:"japanese"`
	Latin       string        `bson:"latin"`
	Translation string        `bson:"translation"`
	Kanji       string        `bson:"kanji"`
	Type        string        `bson:"type"`
	Sets        []any         `bson:"sets"`
	CreatedAt   time.Time     `bson:"createdAt"`
	UpdatedAt   time.Time     `bson:"updatedAt"`
}

type PopulatedWordSchema struct {
	Id          bson.ObjectID `bson:"_id"`
	Japanese    string        `bson:"japanese"`
	Latin       string        `bson:"latin"`
	Translation string        `bson:"translation"`
	Kanji       string        `bson:"kanji"`
	Type        string        `bson:"type"`
	Sets        []SetSchema   `bson:"sets"`
	CreatedAt   time.Time     `bson:"createdAt"`
	UpdatedAt   time.Time     `bson:"updatedAt"`
}

func WordCollection() *mongo.Collection {
	var dbClient *mongo.Client = db.DbConnect()
	var collection = dbClient.Database("tantei").Collection("words")

	return collection
}
